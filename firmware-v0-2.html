HUD Firmware v0.2 - Base Software (RPM, Speed, Greeting, Generic Codes)

Target: ESP32-2432S024 (2.4 inch, 240x320, ILI9341), Arduino IDE, wired CAN via SN65HVD230. Car: 2011 Mercedes E350 (W212, M272). Builds on v0.1.

Read before flashing

  • Pin numbers are PLACEHOLDERS. My web searches failed, so I could not verify your board's pinout. Check the board's pin diagram for two free GPIOs for CAN and the pins for the screen and speaker before wiring or flashing.

  • Screen: install the TFT_eSPI library and use a User_Setup that matches this board. I will give you that file once we confirm pins and touch type.

  • Greeting audio is a placeholder beep. Swap in your own clip later.

  • Power the ESP32 from a 12V to 5V buck converter, never directly from the OBD2 12V line.

  • Test the code reading with the car parked. Reading codes is safe. Do not use this to clear codes or write anything yet.

What this version does

  1. Startup: shows a Welcome screen, plays a beep, and picks a greeting by time of day (time comes from an RTC value later; for now it uses a simple placeholder).

  2. Reads RPM (PID 0C) and speed (PID 0D) back to back for fast updates.

  3. Reads the check engine light status and stored generic codes (Mode 01 PID 01, then Mode 03) every few seconds.

  4. Draws big RPM and speed on screen, with an optional mirror flag for windshield HUD use.

Code

#include <Arduino.h>
#include <TFT_eSPI.h>
#include "driver/twai.h"

TFT_eSPI tft = TFT_eSPI();

// ====== PLACEHOLDERS: CHECK AGAINST YOUR BOARD ======
#define CAN_TX_PIN GPIO_NUM_22
#define CAN_RX_PIN GPIO_NUM_27
#define SPEAKER_PIN 26
// =====================================================

#define MIRROR_FOR_HUD false   // set true for windshield reflection

float rpm = 0;
int speedKmh = 0;
bool milOn = false;
int dtcCount = 0;
String dtcList = "";

unsigned long lastReq = 0, lastCodeCheck = 0;
bool waiting = false;
uint8_t nextPid = 0x0C;   // alternate between RPM and speed

void setupCAN() {
  twai_general_config_t g = TWAI_GENERAL_CONFIG_DEFAULT(CAN_TX_PIN, CAN_RX_PIN, TWAI_MODE_NORMAL);
  twai_timing_config_t t = TWAI_TIMING_CONFIG_500KBITS();
  twai_filter_config_t f = TWAI_FILTER_CONFIG_ACCEPT_ALL();
  twai_driver_install(&g, &t, &f);
  twai_start();
}

void sendRequest(uint8_t mode, uint8_t pid) {
  twai_message_t m = {};
  m.identifier = 0x7DF;
  m.data_length_code = 8;
  if (mode == 0x03) { m.data[0] = 0x01; m.data[1] = 0x03; }
  else { m.data[0] = 0x02; m.data[1] = mode; m.data[2] = pid; }
  twai_transmit(&m, pdMS_TO_TICKS(10));
  lastReq = millis();
  waiting = true;
}

String decodeDTC(uint8_t a, uint8_t b) {
  const char letters[] = {'P', 'C', 'B', 'U'};
  char buf[6];
  buf[0] = letters[(a >> 6) & 0x03];
  buf[1] = '0' + ((a >> 4) & 0x03);
  buf[2] = "0123456789ABCDEF"[a & 0x0F];
  buf[3] = "0123456789ABCDEF"[(b >> 4) & 0x0F];
  buf[4] = "0123456789ABCDEF"[b & 0x0F];
  buf[5] = 0;
  return String(buf);
}

void readReplies() {
  twai_message_t m;
  while (twai_receive(&m, 0) == ESP_OK) {
    if (m.identifier < 0x7E8 || m.identifier > 0x7EF) continue;
    if (m.data[1] == 0x41 && m.data[2] == 0x0C) {
      rpm = ((m.data[3] * 256.0) + m.data[4]) / 4.0; waiting = false;
    } else if (m.data[1] == 0x41 && m.data[2] == 0x0D) {
      speedKmh = m.data[3]; waiting = false;
    } else if (m.data[1] == 0x41 && m.data[2] == 0x01) {
      milOn = m.data[3] & 0x80;
      dtcCount = m.data[3] & 0x7F; waiting = false;
    } else if (m.data[1] == 0x43) {
      // Mode 03 reply: first byte after 0x43 is code count, then 2 bytes per code
      // NOTE: only handles single-frame replies (up to 2 codes). Multi-frame comes next.
      dtcList = "";
      int n = m.data[2];
      for (int i = 0; i < n && i < 2; i++) {
        dtcList += decodeDTC(m.data[3 + i * 2], m.data[4 + i * 2]) + " ";
      }
      waiting = false;
    }
  }
}

void greet() {
  tft.fillScreen(TFT_BLACK);
  tft.setTextColor(TFT_WHITE);
  tft.setTextSize(3);
  tft.setCursor(20, 100);
  tft.println("Welcome,");
  tft.setCursor(20, 140);
  tft.println("Mr. Parsons");
  tone(SPEAKER_PIN, 880, 300);   // placeholder beep, replace with your clip
  delay(2500);
  tft.fillScreen(TFT_BLACK);
}

void drawGauges() {
  tft.setTextSize(6);
  tft.setTextColor(TFT_WHITE, TFT_BLACK);
  tft.setCursor(10, 30);
  tft.printf("%4d", (int)rpm);
  tft.setTextSize(2);
  tft.setCursor(10, 90);
  tft.print("RPM");
  tft.setTextSize(6);
  tft.setCursor(10, 130);
  tft.printf("%3d", (int)(speedKmh * 0.621371));
  tft.setTextSize(2);
  tft.setCursor(10, 190);
  tft.print("MPH");
  tft.setTextSize(2);
  tft.setCursor(10, 220);
  tft.setTextColor(milOn ? TFT_RED : TFT_GREEN, TFT_BLACK);
  tft.printf("CEL: %s  ", milOn ? "ON" : "off");
  tft.setCursor(10, 245);
  tft.print(dtcList + "          ");
}

void setup() {
  Serial.begin(115200);
  tft.init();
  tft.setRotation(MIRROR_FOR_HUD ? 3 : 1);
  // NOTE: true mirroring needs a flipped sprite or the display's MADCTL mirror bit;
  // we will add that when we do the HUD layout.
  pinMode(SPEAKER_PIN, OUTPUT);
  greet();
  setupCAN();
}

void loop() {
  readReplies();
  if (millis() - lastCodeCheck > 5000) {
    sendRequest(0x01, 0x01);              // MIL status and code count
    delay(15);
    if (dtcCount > 0) sendRequest(0x03, 0);  // stored codes
    lastCodeCheck = millis();
  } else if (!waiting || millis() - lastReq > 40) {
    sendRequest(0x01, nextPid);
    nextPid = (nextPid == 0x0C) ? 0x0D : 0x0C;   // alternate RPM and speed
  }
  static unsigned long lastDraw = 0;
  if (millis() - lastDraw > 80) { drawGauges(); lastDraw = millis(); }
}

Known limits of this version

  • Code reads handle only the first two codes in a single frame. Multi-frame (ISO-TP) handling is the next addition.

  • Codes show as raw numbers like P0420. The plain-English description table comes next.

  • The greeting is the same text every time. Time-of-day and weather come after we add the hotspot.

  • True windshield mirroring is not finished. The rotation flag is only a stand-in until the HUD layout step.

  • Speed is converted from km/h to mph on screen.

  • Fuel trims, advanced Mercedes modules, and transmission data are not included yet.

Test order

  1. Flash with no car connected and check the screen and beep work.

  2. Confirm the pins, then wire the CAN transceiver with the car parked.

  3. Watch RPM and speed on screen and on the Serial Monitor.

  4. Only then check code reading.
